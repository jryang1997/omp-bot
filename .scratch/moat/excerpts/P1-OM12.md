   191    resolve(section: string | null | undefined, id: string, answer: "accept" | "reject", expectedRevision?: string): TeamMemoryResolution {
   192      const entries = this.entries(section, true);
   193      const index = entries.findIndex((entry) => entry.id === id);
   194      if (index === -1 || !active(entries[index])) return { claimed: false };
   195      this.checkRevision(entries[index], expectedRevision);
   196      if (entries[index].status === "accepted") return { claimed: true, state: "already_settled" };
   197      if (answer === "accept") {
   198        this.acceptEntry(entries, entries[index]);
   199        this.save();
   200        return { claimed: true, state: "accepted" };
   201      }
   202      entries[index].status = "deleted";
   203      this.save();
   204      return { claimed: true, state: "rejected" };
   205    }

   207    private acceptEntry(entries: TeamMemoryEntry[], proposal: TeamMemoryEntry): void {
   208      const current = entries.filter(previous => previous.status === "accepted" && previous.id !== proposal.id);
   209      if (current.some(previous => previous.kind === proposal.kind && sameName(previous.name, proposal.name) && previous.key !== proposal.key)) {
   210        throw new TeamMemoryConflict("An accepted entry with this kind and name already exists under another key. Use its key or choose a distinct name.");
   211      }
   212      const replaced = current.filter(previous => previous.key === proposal.key);
   213      if (proposal.replaces && (replaced.length !== proposal.replaces.length || replaced.some(previous => !proposal.replaces!.some(base => base.id === previous.id && base.revision === teamMemoryRevision(previous))))) {
   214        throw new TeamMemoryConflict("The accepted fact changed after this proposal. Skip it and create a new proposal against the current fact.");
   215      }
   216      proposal.status = "accepted";
   217      proposal.updatedAt = Date.now();
   218      for (const previous of replaced) {
   219        rememberVersion(previous);
   220        proposal.pinned = proposal.pinned ?? previous.pinned;
   221        proposal.history = [...(proposal.history ?? []), ...(previous.history ?? [])].slice(-50);
   222        previous.status = "superseded";
   223        previous.supersededBy = proposal.id;
   224      }
   225    }
