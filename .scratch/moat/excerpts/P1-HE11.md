   228  def claim_next() -> Optional[dict]:
   229      """Atomically claim one pending send before touching the transport."""
   230      pid = os.getpid()
   231      started = _process_start_time(pid)
   232      with _transaction() as conn:
   233          # created_at carries a DST-varying offset: order by instant, not text.
   234          row = conn.execute(
   235              "SELECT execution_id FROM deliveries WHERE status='pending' "
   236              "ORDER BY julianday(created_at), created_at, execution_id LIMIT 1"
   237          ).fetchone()
   238          if row is None:
   239              return None
   240          cur = conn.execute(
   241              """UPDATE deliveries SET status='delivering', owner_process_id=?,
   242                 owner_pid=?, owner_started_at=?
   243                 WHERE execution_id=? AND status='pending'""",
   244              (_PROCESS_ID, pid, started, row["execution_id"]),
   245          )
   246          if cur.rowcount != 1:
   247              return None
   248          claimed = conn.execute(
   249              "SELECT * FROM deliveries WHERE execution_id=?", (row["execution_id"],)
   250          ).fetchone()
   251          _ACTIVE_DELIVERIES.add(row["execution_id"])
   252      result = dict(claimed)
   253      result["job"] = json.loads(result.pop("job_json"))
   254      return result
