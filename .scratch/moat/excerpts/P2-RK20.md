   354  export function resolveDuplicateEffectGate(
   355    effect: { status: string; result?: unknown },
   356    toolName: string,
   357  ): DuplicateEffectGate {
   358    if (effect.status === "completed") {
   359      return { action: "return", result: effect.result ?? { duplicate: true } };
   360    }
   361    if (effect.status === "denied") {
   362      return {
   363        action: "return",
   364        result: {
   365          error:
   366            "The user denied this action. Do not retry or rephrase it; tell the user and ask what they want instead.",
   367        },
   368      };
   369    }
   370    if (effect.status === "executing") {
   371      return { action: "uncertain", toolName };
   372    }
   373    if (effect.status === "uncertain") {
   374      return { action: "return", result: effect.result ?? uncertainEffectResult(toolName) };
   375    }
   376    if (effect.status === "approved") {
   377      return { action: "execute" };
   378    }
   379    if (effect.status === "intended") {
   380      return { action: "paused" };
   381    }
   382    return { action: "uncertain", toolName };
   383  }
